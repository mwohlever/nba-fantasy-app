function parseArguments(argv) {
  const positional = [];
  const options = {
    round: 1,
    force: false,
  };

  for (
    let index = 0;
    index < argv.length;
    index += 1
  ) {
    const value = argv[index];

    if (value === "--force") {
      options.force = true;
      continue;
    }

    if (
      value === "--round" &&
      argv[index + 1]
    ) {
      options.round = Number(
        argv[index + 1],
      );

      index += 1;
      continue;
    }

    positional.push(value);
  }

  const tournamentId =
    positional[0]?.trim() ?? "";

  if (
    !/^R\d{7}$/i.test(
      tournamentId,
    )
  ) {
    throw new Error(
      [
        "A PGA TOUR tournament ID is required.",
        "",
        "Example:",
        "npm run import-shotcast-course -- R2026013",
      ].join("\n"),
    );
  }

  if (
    !Number.isInteger(options.round) ||
    options.round < 1 ||
    options.round > 4
  ) {
    throw new Error(
      "--round must be an integer from 1 through 4.",
    );
  }

  return {
    tournamentId:
      tournamentId.toUpperCase(),
    round: options.round,
    force: options.force,
  };
}

function loadDotEnvFile(
  filepath,
) {
  if (!existsSync(filepath)) {
    return;
  }

  const lines = readFileSync(
    filepath,
    "utf8",
  ).split(/\r?\n/);

  for (const rawLine of lines) {
    const line = rawLine.trim();

    if (
      !line ||
      line.startsWith("#")
    ) {
      continue;
    }

    const equalsAt =
      line.indexOf("=");

    if (equalsAt < 1) {
      continue;
    }

    const key = line
      .slice(0, equalsAt)
      .trim();

    let value = line
      .slice(equalsAt + 1)
      .trim();

    if (
      (value.startsWith('"') &&
        value.endsWith('"')) ||
      (value.startsWith("'") &&
        value.endsWith("'"))
    ) {
      value = value.slice(1, -1);
    }

    if (
      !process.env[key]
    ) {
      process.env[key] = value;
    }
  }
}

function sleep(milliseconds) {
  return new Promise((resolvePromise) => {
    setTimeout(
      resolvePromise,
      milliseconds,
    );
  });
}

async function fetchWithRetry(
  url,
  init,
  context,
) {
  let lastError = null;

  for (
    let attempt = 1;
    attempt <= MAX_ATTEMPTS;
    attempt += 1
  ) {
    const controller =
      new AbortController();

    const timeout = setTimeout(
      () => controller.abort(),
      REQUEST_TIMEOUT_MS,
    );

    try {
      const response = await fetch(
        url,
        {
          ...init,
          signal: controller.signal,
        },
      );

      if (
        response.ok
      ) {
        return response;
      }

      const responseText =
        await response.text();

      const retryable =
        response.status === 408 ||
        response.status === 429 ||
        response.status >= 500;

      if (
        retryable &&
        attempt < MAX_ATTEMPTS
      ) {
        await sleep(
          400 *
            2 **
              (attempt - 1),
        );

        continue;
      }

      throw new Error(
        `${context} failed (${response.status}): ` +
          responseText.slice(
            0,
            350,
          ),
      );
    } catch (error) {
      lastError = error;

      if (
        attempt >= MAX_ATTEMPTS
      ) {
        break;
      }

      await sleep(
        400 *
          2 **
            (attempt - 1),
      );
    } finally {
      clearTimeout(timeout);
    }
  }

  throw new Error(
    `${context} failed after ${MAX_ATTEMPTS} attempts: ${
      lastError instanceof Error
        ? lastError.message
        : String(lastError)
    }`,
  );
}

async function graphqlRequest(
  operationName,
  query,
  variables,
) {
  const response =
    await fetchWithRetry(
      PGA_GRAPHQL_URL,
      {
        method: "POST",
        headers:
          requestHeaders(),
        body: JSON.stringify({
          operationName,
          query,
          variables,
        }),
      },
      `PGA TOUR ${operationName}`,
    );

  const body =
    await response.json();

  const errors =
    Array.isArray(body.errors)
      ? body.errors
      : [];

  if (
    errors.length > 0
  ) {
    const message = errors
      .map((error) => {
        if (
          error &&
          typeof error ===
            "object" &&
          "message" in error
        ) {
          return String(
            error.message,
          );
        }

        return String(error);
      })
      .join("; ");

    throw new Error(
      `PGA TOUR GraphQL error: ${message}`,
    );
  }

  return body.data ?? {};
}

function normalizeUrl(
  value,
) {
  if (
    typeof value !== "string"
  ) {
    return null;
  }

  const trimmed = value.trim();

  return trimmed || null;
}

function numberOrNull(
  value,
) {
  const numeric =
    Number(value);

  return Number.isFinite(
    numeric,
  )
    ? numeric
    : null;
}

function cleanAsset(
  value,
) {
  if (
    !value ||
    typeof value !== "object"
  ) {
    return null;
  }

  const imageOrg =
    typeof value.imageOrg ===
    "string"
      ? value.imageOrg.trim()
      : "";

  const imagePath =
    typeof value.imagePath ===
    "string"
      ? value.imagePath.trim()
      : "";

  if (
    !imageOrg &&
    !imagePath
  ) {
    return null;
  }

  return {
    imageOrg:
      imageOrg || null,
    imagePath:
      imagePath || null,

    /*
     * The PGA website treats imageOrg as a logical provider.
     * We retain the raw values until the delivery resolver is
     * finalized instead of guessing an invalid direct URL.
     */
    resolvedUrl: null,
  };
}

function cleanCoordinate(
  value,
) {
  if (
    !value ||
    typeof value !== "object"
  ) {
    return null;
  }

  return {
    x: numberOrNull(value.x),
    y: numberOrNull(value.y),
    z: numberOrNull(value.z),
    tourcastX:
      numberOrNull(
        value.tourcastX,
      ),
    tourcastY:
      numberOrNull(
        value.tourcastY,
      ),
    tourcastZ:
      numberOrNull(
        value.tourcastZ,
      ),
  };
}

function cleanPinGreen(
  value,
) {
  if (
    !value ||
    typeof value !== "object"
  ) {
    return null;
  }

  return {
    leftToRight:
      cleanCoordinate(
        value.leftToRightCoords,
      ),
    bottomToTop:
      cleanCoordinate(
        value.bottomToTopCoords,
      ),
  };
}

function extensionFromContentType(
  contentType,
) {
  const normalized =
    contentType
      ?.split(";")[0]
      ?.trim()
      ?.toLowerCase();

  switch (normalized) {
    case "image/png":
      return ".png";

    case "image/webp":
      return ".webp";

    case "image/avif":
      return ".avif";

    case "image/jpeg":
    case "image/jpg":
    default:
      return ".jpg";
  }
}

async function downloadImage(
  url,
  destinationWithoutExtension,
  force,
) {
  if (
    !force
  ) {
    for (const extension of [
      ".jpg",
      ".jpeg",
      ".png",
      ".webp",
      ".avif",
    ]) {
      const candidate =
        destinationWithoutExtension +
        extension;

      if (
        existsSync(candidate) &&
        statSync(candidate).size > 0
      ) {
        return {
          downloaded: false,
          localPath: candidate,
          contentType: null,
          bytes:
            statSync(candidate).size,
        };
      }
    }
  }

  const response =
    await fetchWithRetry(
      url,
      {
        method: "GET",
        headers: {
          Accept: "image/*",
          Referer:
            "https://www.pgatour.com/",
          "User-Agent":
            "111 Sports",
        },
        redirect: "follow",
      },
      `Hole image ${url}`,
    );

  const contentType =
    response.headers.get(
      "content-type",
    );

  if (
    !contentType
      ?.toLowerCase()
      ?.startsWith("image/")
  ) {
    throw new Error(
      `Expected an image from ${url}, received ${contentType ?? "unknown content type"}.`,
    );
  }

  const extension =
    extensionFromContentType(
      contentType,
    );

  const destination =
    destinationWithoutExtension +
    extension;

  const temporary =
    destination + ".download";

  mkdirSync(
    dirname(destination),
    {
      recursive: true,
    },
  );

  rmSync(
    temporary,
    {
      force: true,
    },
  );

  await pipeline(
    response.body,
    createWriteStream(
      temporary,
    ),
  );

  if (
    !existsSync(temporary) ||
    statSync(temporary).size === 0
  ) {
    throw new Error(
      `Downloaded image was empty: ${url}`,
    );
  }

  renameSync(
    temporary,
    destination,
  );

  return {
    downloaded: true,
    localPath: destination,
    contentType,
    bytes:
      statSync(destination).size,
  };
}

function publicPathFromFile(
  filepath,
) {
  const publicRoot =
    resolve(
      process.cwd(),
      "public",
    );

  const absolute =
    resolve(filepath);

  if (
    !absolute.startsWith(
      publicRoot,
    )
  ) {
    throw new Error(
      `File is outside public/: ${absolute}`,
    );
  }

  return (
    "/" +
    absolute
      .slice(
        publicRoot.length,
      )
      .replaceAll("\\", "/")
      .replace(/^\/+/, "")
  );
}
